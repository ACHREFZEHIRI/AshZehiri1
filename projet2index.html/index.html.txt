      if(!h) continue;
      const isCount = COUNT_KEYWORDS.test(h);
      const periodKey = extractPeriodKey(h) || (isCount ? '__count_total__' : h);
      cols.push({index:i, header:h, type: isCount ? 'count' : 'amount', periodKey});
    }
    if(!cols.length) throw new Error("Aucune colonne de données détectée après la colonne fournisseur.");

    const hasCountData = cols.some(c=>c.type==='count');

    // ordered unique period keys, based on amount columns primarily
    const periodOrder = [];
    cols.forEach(c=>{
      if(c.type==='amount' && periodOrder.indexOf(c.periodKey)===-1) periodOrder.push(c.periodKey);
    });

    const suppliers = [];
    for(let r=1;r<rows.length;r++){
      const row = rows[r];
      const name = String(row[0]||'').trim();
      if(!name) continue;
      const months = {};
      periodOrder.forEach(pk=>{ months[pk] = {amount:0, count:0}; });
      let countNoPeriod = 0;

      cols.forEach(c=>{
        const raw = row[c.index];
        const val = parseNumber(raw);
        if(c.type === 'amount'){
          months[c.periodKey].amount += val;
        } else {
          if(months[c.periodKey]) months[c.periodKey].count += val;
          else countNoPeriod += val;
        }
      });

      const totalAmount = periodOrder.reduce((s,pk)=>s+months[pk].amount,0);
      const totalCount = periodOrder.reduce((s,pk)=>s+months[pk].count,0) + countNoPeriod;

      if(totalAmount === 0 && totalCount === 0) continue; // skip fully empty rows
      suppliers.push({name, months, totalAmount, totalCount});
    }

    if(!suppliers.length) throw new Error('Aucune ligne fournisseur valide trouvée.');

    return {
      kind:'amounts',
      fileName,
      sourceFiles:[fileName],
      importedAt: new Date().toISOString(),
      periodKeys: periodOrder,
      periodLabels: periodOrder.map(periodLabelFromKey),
      hasCountData,
      suppliers
    };
  }

  function findColIndex(header, re){
    for(let i=0;i<header.length;i++){ if(re.test(String(header[i]||''))) return i; }
    return -1;
  }

  const ORDERS_REF_RE = /r[ée]f[ée]rence|order\s*ref/i;
  const ORDERS_DATE_RE = /date/i;
  const ORDERS_SUPPLIER_RE = /fournisseur|vendor|supplier/i;
  const ORDERS_TOTAL_RE = /^total$|montant|amount/i;

  // A file is treated as a "purchase order log" (one row per order, possibly interleaved
  // with group/subtotal rows from an ERP export) when its header exposes at least a
  // reference/date/supplier column. Otherwise it's treated as the wide amounts format.
  function detectFileType(header){
    const hasRef = findColIndex(header, ORDERS_REF_RE) > -1;
    const hasDate = findColIndex(header, ORDERS_DATE_RE) > -1;
    const hasSupplier = findColIndex(header, ORDERS_SUPPLIER_RE) > -1;
    return (hasDate && hasSupplier) ? 'orders' : 'amounts';
  }

  function parseOrdersWorkbook(workbook, fileName){
    const sheetName = workbook.SheetNames[0];
    const sheet = workbook.Sheets[sheetName];
    const rows = XLSX.utils.sheet_to_json(sheet, {header:1, raw:false, dateNF:'yyyy-mm-dd', defval:''});
    if(!rows.length) throw new Error('Le fichier semble vide.');

    const header = rows[0].map(h=>String(h||'').trim());
    const dateIdx = findColIndex(header, ORDERS_DATE_RE);
    const supIdx = findColIndex(header, ORDERS_SUPPLIER_RE);
    const totalIdx = findColIndex(header, ORDERS_TOTAL_RE);
    if(dateIdx === -1 || supIdx === -1) throw new Error("Colonnes 'date' et 'fournisseur' introuvables dans ce fichier.");

    // Subtotal/group rows (common in ERP exports grouped by supplier/year/month) have a
    // blank date or blank supplier — real order lines always have both, so filtering on
    // that is enough to skip them without needing to recognize a specific reference format.
    const bySupplier = {}; // normKey -> {displayName, periods:{pk:{count, amount}}}
    let rowsUsed = 0;

    for(let r=1;r<rows.length;r++){
      const row = rows[r];
      const dateRaw = row[dateIdx];
      const supRaw = row[supIdx];
      if(!dateRaw || !supRaw) continue;
      const m = /^(\d{4})-(\d{2})/.exec(String(dateRaw));
      if(!m) continue;
      const pk = m[1] + '-' + m[2];

      const displayName = String(supRaw).replace(/\t/g,' ').replace(/\s+/g,' ').trim();
      if(!displayName) continue;
      const normKey = normalize(displayName);

      if(!bySupplier[normKey]) bySupplier[normKey] = {displayName, periods:{}};
      const bucket = bySupplier[normKey];
      if(!bucket.periods[pk]) bucket.periods[pk] = {count:0, amount:0};
      bucket.periods[pk].count += 1;
      if(totalIdx > -1) bucket.periods[pk].amount += parseNumber(row[totalIdx]);
      rowsUsed++;
    }

    if(!rowsUsed) throw new Error("Aucune ligne de commande valide (date + fournisseur) trouvée dans ce fichier.");

    const periodSet = new Set();
    Object.values(bySupplier).forEach(b=>Object.keys(b.periods).forEach(pk=>periodSet.add(pk)));
    const periodOrder = Array.from(periodSet).sort();

    const suppliers = Object.keys(bySupplier).map(normKey=>{
      const b = bySupplier[normKey];
      const months = {};
      let totalCount = 0, totalAmount = 0;
      periodOrder.forEach(pk=>{
        const p = b.periods[pk] || {count:0, amount:0};
        months[pk] = {amount:p.amount, count:p.count};
        totalCount += p.count; totalAmount += p.amount;
      });
      return {name:b.displayName, normKey, months, totalAmount, totalCount};
    });

    return {
      kind:'orders',
      fileName,
      sourceFiles:[fileName],
      importedAt: new Date().toISOString(),
      periodKeys: periodOrder,
      periodLabels: periodOrder.map(periodLabelFromKey),
      rowsUsed,
      suppliers
    };
  }

  // Merges a newly-imported source into whatever was previously stored for that slot so
  // successive monthly imports accumulate into one running history instead of replacing it.
  // Periods covered by the new file overwrite the old values for those periods (in case a
  // month's figures were corrected); periods only present in the old file are kept untouched.
  function mergeSource(oldSource, newSource){
    if(!oldSource) return newSource;

    const periodSet = new Set([...oldSource.periodKeys, ...newSource.periodKeys]);
    const periodKeys = Array.from(periodSet).sort();
    const newPeriodSet = new Set(newSource.periodKeys);

    const supplierMap = {}; // normKey -> {name, normKey, months:{pk:{amount,count}}}
    function ensure(normKey, displayName){
      if(!supplierMap[normKey]) supplierMap[normKey] = {name:displayName, normKey, months:{}};
      return supplierMap[normKey];
    }

    oldSource.suppliers.forEach(s=>{
      const normKey = s.normKey || normalize(s.name);
      const b = ensure(normKey, s.name);
      oldSource.periodKeys.forEach(pk=>{
        if(newPeriodSet.has(pk)) return; // this period is authoritatively replaced by newSource below
        b.months[pk] = {amount:s.months[pk].amount, count:s.months[pk].count};
      });
    });
    newSource.suppliers.forEach(s=>{
      const normKey = s.normKey || normalize(s.name);
      const b = ensure(normKey, s.name); // newer display name wins
      b.name = s.name;
      newSource.periodKeys.forEach(pk=>{ b.months[pk] = {amount:s.months[pk].amount, count:s.months[pk].count}; }); // overwrite this period
    });

    const suppliers = Object.keys(supplierMap).map(k=>{
      const b = supplierMap[k];
      let totalAmount=0, totalCount=0;
      periodKeys.forEach(pk=>{
        const m = b.months[pk] || {amount:0, count:0};
        b.months[pk] = m;
        totalAmount += m.amount; totalCount += m.count;
      });
      return {name:b.name, normKey:b.normKey, months:b.months, totalAmount, totalCount};
    });

    const sourceFiles = Array.from(new Set([...(oldSource.sourceFiles||[oldSource.fileName]), ...(newSource.sourceFiles||[newSource.fileName])]));

    return {
      kind: newSource.kind,
      fileName: newSource.fileName,
      sourceFiles,
      importedAt: new Date().toISOString(),
      periodKeys,
      periodLabels: periodKeys.map(periodLabelFromKey),
      hasCountData: newSource.kind==='amounts' ? (!!oldSource.hasCountData || !!newSource.hasCountData) : undefined,
      rowsUsed: newSource.kind==='orders' ? suppliers.reduce((s,x)=>s+x.totalCount,0) : undefined,
      suppliers
    };
  }

  // Merges whichever source(s) are loaded into the single shape every render function expects.
  // Amounts take priority for any period they cover; the orders log fills in order counts
  // everywhere, and also supplies amount totals for periods the amounts file doesn't cover
  // (e.g. more recent months only present in the order log).
  function buildDataset(){
    if(!amountsSource && !ordersSource) return null;

    const periodSet = new Set();
    (amountsSource ? amountsSource.periodKeys : []).forEach(pk=>periodSet.add(pk));
    (ordersSource ? ordersSource.periodKeys : []).forEach(pk=>periodSet.add(pk));
    const periodKeys = Array.from(periodSet).sort();
    const periodLabels = periodKeys.map(periodLabelFromKey);
    const amountsPeriodSet = new Set(amountsSource ? amountsSource.periodKeys : []);

    const supplierMap = {}; // normKey -> {name, months:{pk:{amount,count}}}
    function ensureSupplier(normKey, displayName){
      if(!supplierMap[normKey]) supplierMap[normKey] = {name:displayName, months:{}};
      periodKeys.forEach(pk=>{ if(!supplierMap[normKey].months[pk]) supplierMap[normKey].months[pk] = {amount:0, count:0}; });
      return supplierMap[normKey];
    }

    if(amountsSource){
      amountsSource.suppliers.forEach(s=>{
        const normKey = normalize(s.name);
        const bucket = ensureSupplier(normKey, s.name);
        amountsSource.periodKeys.forEach(pk=>{
          bucket.months[pk].amount += s.months[pk].amount;
          bucket.months[pk].count += s.months[pk].count; // in case the amounts file itself had a count column
        });
      });
    }
    if(ordersSource){
      ordersSource.suppliers.forEach(s=>{
        const normKey = s.normKey;
        const existingName = supplierMap[normKey] ? supplierMap[normKey].name : s.name;
        const bucket = ensureSupplier(normKey, existingName);
        ordersSource.periodKeys.forEach(pk=>{
          bucket.months[pk].count += s.months[pk].count;
          if(!amountsPeriodSet.has(pk)) bucket.months[pk].amount += s.months[pk].amount;
        });
      });
    }

    const suppliers = Object.keys(supplierMap).map(k=>{
      const b = supplierMap[k];
      const totalAmount = periodKeys.reduce((s,pk)=>s+b.months[pk].amount,0);
      const totalCount = periodKeys.reduce((s,pk)=>s+b.months[pk].count,0);
      return {name:b.name, months:b.months, totalAmount, totalCount};
    }).filter(s=>s.totalAmount>0 || s.totalCount>0);

    return {
      periodKeys,
      periodLabels,
      hasCountData: !!ordersSource || (!!amountsSource && amountsSource.hasCountData),
      suppliers
    };
  }

  function onFileSelected(e){
    const file = e.target.files[0];
    if(!file) return;
    const reader = new FileReader();
    reader.onload = function(evt){
      try{
        const data = new Uint8Array(evt.target.result);
        const workbook = XLSX.read(data, {type:'array'});
        const sheet = workbook.Sheets[workbook.SheetNames[0]];
        const headerRow = (XLSX.utils.sheet_to_json(sheet, {header:1, raw:false, range:0})[0] || []).map(h=>String(h||'').trim());
        const type = detectFileType(headerRow);

        if(type === 'orders'){
          ordersSource = mergeSource(ordersSource, parseOrdersWorkbook(workbook, file.name));
          saveSource(STORAGE_KEY_ORDERS, ordersSource);
        } else {
          amountsSource = mergeSource(amountsSource, parseAmountsWorkbook(workbook, file.name));
          saveSource(STORAGE_KEY_AMOUNTS, amountsSource);
        }
        dataset = buildDataset();
        renderAll();
      } catch(err){
        alert("Impossible de lire ce fichier : " + err.message);
      }
      fileInput.value = '';
    };
    reader.onerror = function(){
      alert("Erreur de lecture du fichier.");
      fileInput.value = '';
    };
    reader.readAsArrayBuffer(file);
  }

    try{
      const res = await window.storage.get(STORAGE_KEY_AMOUNTS, false);

      if(res && res.value) ordersSource = JSON.parse(res.value);
    } catch(err){ /* nothing stored yet */ }
    dataset = buildDataset();

  }

  // Renders one chart in isolation: if it throws (bad data, library not loaded, etc.)
  // the error is shown inline in that chart's box instead of silently leaving it blank
  // and instead of aborting the other charts still queued to render.
  function safeRender(fn, canvasId){
    try{
      fn();
    } catch(err){
      console.error('Erreur de rendu (' + canvasId + ')', err);
      const box = $('#' + canvasId).closest('.chart-box');
      if(box) box.innerHTML = '<div style="color:var(--bad); font-family:\'IBM Plex Mono\',monospace; font-size:12px; padding:12px;">Erreur d\'affichage : ' + escapeHtml(err.message) + '</div>';
    }
  }

  function renderSourcesStatus(){
    const chip = (label, source)=>{
      if(!source){
        return `<div class="source-chip"><span class="kind missing">${label} — aucun fichier</span></div>`;
      }

    };

  }

  // Linear regression slope of a series (index = x, value = y)
  function linRegSlope(vals){
    const n = vals.length;
    if(n < 2) return 0;
    const xMean = (n-1)/2;
    const yMean = vals.reduce((a,b)=>a+b,0)/n;
    let num=0, den=0;
    for(let i=0;i<n;i++){ num += (i-xMean)*(vals[i]-yMean); den += (i-xMean)*(i-xMean); }
    return den === 0 ? 0 : num/den;
  }

  function classifyGrowth(pct){
    if(pct === null) return 'flat';
    if(pct > 15) return 'up';
    if(pct < -15) return 'down';
    return 'flat';
  }
  function classifyRegularity(pct){
    if(pct >= 80) return 'regular';
    if(pct >= 40) return 'irregular';
    return 'occasional';
  }
  function scoreLabel(score){
    if(score >= 70) return {cls:'strategic', text:'Stratégique'};
    if(score >= 45) return {cls:'reliable', text:'Fiable'};
    if(score >= 25) return {cls:'watch', text:'À surveiller'};
    return {cls:'weak', text:'Occasionnel'};
  }

  // Computes, once per dataset, a per-supplier evaluation: regularity (%), growth trend (%),
  // and a 0-100 composite score blending purchase volume, order regularity and growth momentum.
  function computeSupplierMetrics(){
    if(!dataset) return [];
    const nPeriods = dataset.periodKeys.length;
    const maxTotal = Math.max(...dataset.suppliers.map(s=>s.totalAmount), 1);
    const grandTotal = dataset.suppliers.reduce((s,x)=>s+x.totalAmount,0) || 1;

    return dataset.suppliers.map(s=>{
      const vals = dataset.periodKeys.map(pk=>s.months[pk].amount);
      const activeMonths = vals.filter(v=>v>0).length;
      const regularity = nPeriods ? (activeMonths/nPeriods)*100 : 0;

      const avgVal = vals.reduce((a,b)=>a+b,0)/Math.max(vals.length,1);
      let growth = null;
      if(avgVal > 0 && vals.length >= 2){
        const slope = linRegSlope(vals);
        growth = (slope*(vals.length-1)/avgVal)*100;
      }

      const volumeScore = s.totalAmount>0 ? Math.min(1, Math.log(1+s.totalAmount)/Math.log(1+maxTotal))*40 : 0;
      const regScore = (regularity/100)*30;
      const growthClamped = growth===null ? 0 : Math.max(-100, Math.min(100, growth));
      const growthScore = growth===null ? 15 : ((growthClamped+100)/200)*30;
      const score = Math.round(volumeScore + regScore + growthScore);

      return {
        name: s.name,
        total: s.totalAmount,
        share: (s.totalAmount/grandTotal)*100,
        count: s.totalCount,
        regularity,
        growth,
        score,
        months: s.months
      };
    });
  }

  function monthlyTotals(){
    return dataset.periodKeys.map(pk=>{
      let amount=0, count=0;
      dataset.suppliers.forEach(s=>{ amount += s.months[pk].amount; count += s.months[pk].count; });
      return {amount, count};
    });
  }

  function renderTickets(){
    const totals = monthlyTotals();
    const grandTotal = dataset.suppliers.reduce((s,x)=>s+x.totalAmount,0);
    const activeSuppliers = dataset.suppliers.filter(s=>s.totalAmount>0).length;
    const totalCount = dataset.suppliers.reduce((s,x)=>s+x.totalCount,0);

    let bestIdx = 0;
    totals.forEach((t,i)=>{ if(t.amount > totals[bestIdx].amount) bestIdx = i; });
    const bestMonthLabel = dataset.periodLabels[bestIdx];
    const bestMonthValue = totals[bestIdx].amount;

    let evolPct = null;
    if(totals.length >= 2){
      const prev = totals[totals.length-2].amount;
      const last = totals[totals.length-1].amount;
      if(prev > 0) evolPct = ((last-prev)/prev)*100;
    }

    const tickets = [
      {
        eyebrow:'Achats totaux', value: fmtTND(grandTotal),
        foot: dataset.periodLabels.length + ' mois suivis',
        stamp: evolPct===null ? null : (evolPct>=0 ? 'up' : 'down'),
        stampText: evolPct===null ? '' : (evolPct>=0?'▲':'▼') + Math.abs(evolPct).toFixed(0) + '%'
      },
      {
        eyebrow:'Fournisseurs actifs', value: activeSuppliers + ' <small>/ ' + dataset.suppliers.length + '</small>',
        foot: 'ayant généré un achat sur la période'
      },
      {
        eyebrow:'Commandes enregistrées',
        value: dataset.hasCountData ? totalCount.toLocaleString('fr-FR') : '<span class="dash">—</span>',
        foot: dataset.hasCountData ? 'total sur la période' : "ajoutez une colonne 'Commandes' pour ce suivi"
      },
      {
        eyebrow:'Meilleur mois', value: fmtTND(bestMonthValue),
        foot: bestMonthLabel
      },
      {
        eyebrow:'Fournisseurs réguliers',
        value: metricsCache.filter(m=>m.regularity>=80).length + ' <small>/ ' + dataset.suppliers.length + '</small>',
        foot: 'actifs sur au moins 80% des mois'
      },
      {
        eyebrow:'Score moyen',
        value: (metricsCache.reduce((s,m)=>s+m.score,0)/Math.max(metricsCache.length,1)).toFixed(0) + ' <small>/ 100</small>',
        foot: 'volume + régularité + croissance'
      }
    ];

    $('#tickets').innerHTML = tickets.map(t=>`
      <div class="ticket">
        ${t.stamp ? `<div class="stamp ${t.stamp==='down'?'down':''}">${t.stampText}</div>` : ''}
        <div class="eyebrow">${t.eyebrow}</div>
        <div class="value">${t.value}</div>
        <div class="foot">${t.foot}</div>
      </div>
    `).join('');
  }

      name: m.name, score: m.score, growth: m.growth
    }));

    const avgShare = points.length ? points.reduce((s,p)=>s+p.y,0)/points.length : 0;

    matrixChartInstance = new Chart(ctx, {
      type:'bubble',
      data:{ datasets:[{
        data: points,
        backgroundColor: points.map(p=>colorFor(p.growth) + '99'),
        borderColor: points.map(p=>colorFor(p.growth)),
        borderWidth:1.5
      }]},
      options:{
        responsive:true, maintainAspectRatio:false,
        plugins:{
          legend:{display:false},
          quadrantDivider:{xMid:50, yMid:avgShare},
          tooltip:{callbacks:{
            title:(items)=> items[0].raw.name,
            label:(c)=> 'Régularité ' + c.raw.x.toFixed(0) + '% · Part ' + c.raw.y.toFixed(1) + '% · Score ' + c.raw.score
          }}
        },
        scales:{
          x:{ title:{display:true, text:'Régularité des commandes (%)', color:'#6B6F73', font:{family:'IBM Plex Mono', size:10}},
              min:0, max:100, ticks:{color:'#6B6F73', font:{family:'IBM Plex Mono', size:10}}, grid:{color:'rgba(33,30,26,0.08)'} },
          y:{ title:{display:true, text:'Part des achats (%)', color:'#6B6F73', font:{family:'IBM Plex Mono', size:10}},
              beginAtZero:true, ticks:{color:'#6B6F73', font:{family:'IBM Plex Mono', size:10}}, grid:{color:'rgba(33,30,26,0.08)'} }
        }
      }
    });
  }

  function sparkPath(months, w, h){
    const vals = dataset.periodKeys.map(pk=>months[pk].amount);
    const max = Math.max(...vals, 1);
    const step = w / Math.max(vals.length-1,1);
    const pts = vals.map((v,i)=>{
      const x = i*step;
      const y = h - (v/max)*h;
      return x.toFixed(1)+','+y.toFixed(1);
    });
    return pts.join(' ');
  }

  function renderTable(){
    if(!dataset) return;
    const q = normalize($('#searchInput').value);

    let rows = metricsCache.slice();

    if(q){
      rows = rows.filter(r=>normalize(r.name).indexOf(q) > -1);
    }

    rows.sort((a,b)=>{
      let av=a[sortState.key], bv=b[sortState.key];
      if(sortState.key==='name'){ av = normalize(av); bv = normalize(bv); }
      if(sortState.key==='growth'){ av = av===null?-Infinity:av; bv = bv===null?-Infinity:bv; }
      if(av < bv) return sortState.dir==='asc' ? -1 : 1;
      if(av > bv) return sortState.dir==='asc' ? 1 : -1;
      return 0;
    });

    document.querySelectorAll('thead th[data-key]').forEach(th=>{
      th.classList.toggle('sorted', th.dataset.key===sortState.key);
      th.innerHTML = th.textContent.replace(/\s*[▲▼]$/,'') + (th.dataset.key===sortState.key ? ` <span class="arrow">${sortState.dir==='asc'?'▲':'▼'}</span>` : '');
    });

    const maxTotal = Math.max(...rows.map(r=>r.total), 1);

    $('#tableBody').innerHTML = rows.map(r=>{
      const growthClass = classifyGrowth(r.growth);
      const growthText = r.growth===null ? '<span class="dash">—</span>' : fmtPct(r.growth, true);
      const regClass = classifyRegularity(r.regularity);
      const spark = sparkPath(r.months, 90, 24);
      const countText = dataset.hasCountData ? r.count.toLocaleString('fr-FR') : '<span class="dash">—</span>';
      const label = scoreLabel(r.score);
      return `<tr>
        <td class="name">${escapeHtml(r.name)}</td>
        <td class="num">${fmtTND(r.total)}</td>
        <td class="num">
          <div class="bar-cell">
            <span>${r.share.toFixed(1)}%</span>
            <div class="mini-bar-track"><div class="mini-bar-fill" style="width:${(r.total/maxTotal*100).toFixed(1)}%"></div></div>
          </div>
        </td>
        <td class="num">${countText}</td>
        <td class="num"><span class="reg-pill ${regClass}">${r.regularity.toFixed(0)}%</span></td>
        <td class="num trend ${growthClass}">${growthText}</td>
        <td class="num"><span class="badge ${label.cls}"><span class="dot"></span>${r.score} · ${label.text}</span></td>
        <td>
          <svg class="spark" width="90" height="24" viewBox="0 0 90 24" preserveAspectRatio="none">
            <polyline points="${spark}" fill="none" stroke="#E2841E" stroke-width="2" vector-effect="non-scaling-stroke"/>
          </svg>
        </td>
      </tr>`;
    }).join('') || `<tr><td colspan="8" style="text-align:center;color:var(--muted);padding:24px;">Aucun fournisseur ne correspond à votre recherche.</td></tr>`;
  }

  function escapeHtml(s){
    return String(s).replace(/[&<>"']/g, c=>({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
  }

    // init (inside startApp)
    renderAll();
    loadSources();
  } // end startApp

  bootWhenReady();
